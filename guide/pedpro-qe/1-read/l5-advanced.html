L5.1Mutation Testing ⭐⭐

metric เดียวที่วัด “ความแข็งแรงของ oracle” ได้เป็นตัวเลข

coverage บอกว่า “โค้ดถูกรัน” · mutation score บอกว่า “ถ้าโค้ดผิด เทสจะรู้มั้ย”
และมันสำคัญเป็นพิเศษกับคุณ เพราะ oracle ที่อ่อนคือ failure mode หลักของเทสที่ agent เขียน (L4.12 P5)

หลักการ — และความเชื่อมโยงกับ RIP model

1. เอาโค้ด production มาแก้ทีละจุดเล็กๆ ให้ "ผิด" → เรียกว่า mutant if (x >= 100) → if (x > 100) 2. รันเทสทั้งชุดกับ mutant 3. เทสแดง → "killed" ✓ เทสจับได้ เทสเขียว → "survived" ✗ เทสจับไม่ได้ mutation score = killed / (mutants ทั้งหมด − equivalent)

นี่คือ RIP model ที่ถูกทำให้เป็นเครื่องมือ — mutant ที่รอดคือ defect ที่ reach ได้ infect ได้ แต่ไม่ propagate ถึง assertion

#!/usr/bin/env node
import { readFileSync, writeFileSync } from "node:fs"

const report = JSON.parse(readFileSync("reports/mutation.json", "utf8"))

// 1. จำแนกอัตโนมัติเท่าที่ทำได้
function classify(m, fileSrc) {
  if (m.status === "NoCoverage") return "M1"
  if (m.status !== "Survived") return null

  const line = fileSrc.split("\n")[m.location.start.line - 1] ?? ""
  if (/console\.|logger\.|\/\/|\/\*/.test(line)) return "M5"
  // equivalent ที่พบบ่อย: การเปลี่ยน <= เป็น < ในลูปที่ไม่มีผล
  if (m.mutatorName === "EqualityOperator" && /\.length/.test(line)) return "M3?"
  return "M2"       // ⭐ default เป็น M2 — ต้องพิสูจน์ว่าไม่ใช่
}

// 2. ⭐ clustering — mutant หลายตัวในฟังก์ชันเดียวกัน = ปัญหาเดียว
const clusters = new Map()
for (const file of report.files) {
  const src = readFileSync(file.name, "utf8")
  for (const m of file.mutants) {
    const kind = classify(m, src)
    if (!kind || kind === "M5" || kind === "M3?") continue
    const fn = findEnclosingFunction(src, m.location.start.line)
    const key = `${file.name}::${fn}`
    if (!clusters.has(key)) clusters.set(key, { kind, mutants: [], fn, file: file.name })
    clusters.get(key).mutants.push(m)
  }
}

// 3. เรียงตามจำนวน mutant ที่รอดในแต่ละจุด
const sorted = [...clusters.values()].sort((a, b) => b.mutants.length - a.mutants.length)

console.log(`mutant ที่ต้องจัดการ ${report.files.reduce((s,f)=>s+f.mutants.length,0)} ตัว`)
console.log(`→ ยุบเหลือ ${sorted.length} จุดที่ต้องแก้\n`)

for (const c of sorted.slice(0, 20)) {
  console.log(`  [${c.kind}] ${c.file} :: ${c.fn}()  — mutant รอด ${c.mutants.length} ตัว`)
  if (c.mutants.length >= 5)
    console.log(`      ⭐ รอดเยอะขนาดนี้ = ฟังก์ชันนี้แทบไม่มี oracle เลย`)
}

writeFileSync("reports/mutant-clusters.json", JSON.stringify(sorted, null, 2))
Clustering เปลี่ยนงานจาก “ฆ่า 612 ตัว” เป็น “แก้ 40 จุด”

และจุดที่มี mutant รอดเกิน 5 ตัวคือสัญญาณของ design problem ไม่ใช่แค่ปัญหาเทส — ฟังก์ชันที่ mutate ยังไงก็ไม่มีเทสจับ มักเป็นฟังก์ชันที่ทำหลายอย่างเกินไปจนไม่มีใครเขียน assertion ที่คมได้

Custom Mutator ของโดเมน ⭐⭐

mutator มาตรฐาน (เปลี่ยน + เป็น -, >= เป็น >) จับ off-by-one ได้ดี แต่ไม่จับ defect ที่เป็นลักษณะเฉพาะของระบบคุณ

Mutatorทำอะไรถ้ารอด แปลว่า
DoAdminRemovalลบ DoAdmin/DoIp ออกจาก payload⭐ ไม่มีเทสตรวจ audit trail
AwaitRemovalลบ await ออกจาก async call⭐⭐ race condition ที่ไม่มีใครจับ
TimezoneShiftเปลี่ยน Asia/Bangkok เป็น UTCไม่มีเทสขอบวัน 07:00
ThaiCollationเปลี่ยน localeCompare(a,b,"th-TH") เป็น <ไม่มีเทสการเรียงไทย
IdempotencyBypassลบการเช็ค idempotency key⭐⭐ ไม่มีเทสยิงซ้ำ
BuddhistYearเปลี่ยน +543 เป็น -543ไม่มีเทส พ.ศ./ค.ศ.
MoneyRoundingเปลี่ยน round ↔ floor ↔ ceilไม่มีเทสทศนิยม
AuthzSkipเปลี่ยนการเช็คสิทธิ์เป็น true⭐⭐ authz matrix ไม่ครอบ
typescriptmutation/custom-mutators.ts — ทำเองด้วย ts-morph เพราะ Stryker ยังไม่รองรับ plugin แบบนี้
import { Project, SyntaxKind } from "ts-morph"

export type CustomMutant = {
  id: string
  file: string
  line: number
  mutator: string
  original: string
  mutated: string
  apply: () => void
  revert: () => void
}

export function generateCustomMutants(globs: string[]): CustomMutant[] {
  const project = new Project({ tsConfigFilePath: "tsconfig.json" })
  const mutants: CustomMutant[] = []

  for (const sf of project.getSourceFiles(globs)) {
    // ── AwaitRemoval ⭐⭐ ──
    sf.forEachDescendant((node) => {
      if (node.getKind() !== SyntaxKind.AwaitExpression) return
      const expr = node.asKindOrThrow(SyntaxKind.AwaitExpression)
      const inner = expr.getExpression().getText()
      // ⭐ เฉพาะ call ที่มีผลข้างเคียง — ไม่งั้นได้ equivalent mutant เต็มไปหมด
      if (!/(save|insert|update|delete|credit|debit|post|commit)/i.test(inner)) return
      mutants.push({
        id: `AWAIT-${sf.getBaseName()}-${expr.getStartLineNumber()}`,
        file: sf.getFilePath(), line: expr.getStartLineNumber(),
        mutator: "AwaitRemoval",
        original: expr.getText(), mutated: inner,
        apply: () => expr.replaceWithText(inner),
        revert: () => {},          // ใช้ git checkout แทน
      })
    })

    // ── DoAdminRemoval ──
    sf.forEachDescendant((node) => {
      if (node.getKind() !== SyntaxKind.PropertyAssignment) return
      const prop = node.asKindOrThrow(SyntaxKind.PropertyAssignment)
      if (!["DoAdmin", "DoIp"].includes(prop.getName())) return
      mutants.push({
        id: `DOADMIN-${sf.getBaseName()}-${prop.getStartLineNumber()}`,
        file: sf.getFilePath(), line: prop.getStartLineNumber(),
        mutator: "DoAdminRemoval",
        original: prop.getText(), mutated: "",
        apply: () => prop.remove(),
        revert: () => {},
      })
    })

    // ── IdempotencyBypass ──
    sf.forEachDescendant((node) => {
      if (node.getKind() !== SyntaxKind.IfStatement) return
      const stmt = node.asKindOrThrow(SyntaxKind.IfStatement)
      const cond = stmt.getExpression().getText()
      if (!/idempoten|alreadyProcessed|isDuplicate|existingTxn/i.test(cond)) return
      mutants.push({
        id: `IDEM-${sf.getBaseName()}-${stmt.getStartLineNumber()}`,
        file: sf.getFilePath(), line: stmt.getStartLineNumber(),
        mutator: "IdempotencyBypass",
        original: cond, mutated: "false",
        apply: () => stmt.getExpression().replaceWithText("false"),
        revert: () => {},
      })
    })
  }
  return mutants
}

// Stryker บอกว่า mutant แต่ละตัวถูกฆ่าโดยเทสตัวไหน
// → กลับข้าง: เทสแต่ละตัวฆ่า mutant กี่ตัว

const report = JSON.parse(readFileSync("reports/mutation.json", "utf8"))
const killsByTest = new Map()

for (const file of report.files)
  for (const m of file.mutants)
    if (m.status === "Killed")
      for (const t of m.killedBy ?? [])
        killsByTest.set(t, (killsByTest.get(t) ?? new Set()).add(m.id))

const tests = [...killsByTest.entries()]
  .map(([test, kills]) => ({ test, kills: kills.size, mutantIds: kills }))
  .sort((a, b) => b.kills - a.kills)

// ── หมวดที่ 1: เทสที่ไร้ค่า ──
const worthless = report.tests.filter((t) => !killsByTest.has(t.id))
console.log(`\n❌ เทสที่ไม่ฆ่า mutant เลย (${worthless.length} ตัว):`)
worthless.slice(0, 20).forEach((t) => console.log(`   ${t.name}`))
console.log("   ⭐ อาจเป็นเทสที่ไม่มี assertion ที่มีความหมาย — ตรวจก่อนลบ")

// ── หมวดที่ 2: เทสที่ซ้ำซ้อน ──
const redundant = []
for (let i = 0; i < tests.length; i++)
  for (let j = i + 1; j < tests.length; j++) {
    const a = tests[i].mutantIds, b = tests[j].mutantIds
    if ([...b].every((id) => a.has(id)))
      redundant.push({ superset: tests[i].test, subset: tests[j].test })
  }
console.log(`\n⚠️  เทสที่ซ้ำซ้อน (${redundant.length} คู่) — U = Unique ใน F.I.R.S.T+U`)

// ── หมวดที่ 3: ⭐ smoke suite ที่คัดด้วย greedy set cover ──
const allKilled = new Set(tests.flatMap((t) => [...t.mutantIds]))
const covered = new Set()
const smoke = []
while (covered.size < allKilled.size && smoke.length < 25) {
  const best = tests
    .filter((t) => !smoke.includes(t.test))
    .map((t) => ({ t, gain: [...t.mutantIds].filter((id) => !covered.has(id)).length }))
    .sort((a, b) => b.gain - a.gain)[0]
  if (!best || best.gain === 0) break
  smoke.push(best.t.test)
  best.t.mutantIds.forEach((id) => covered.add(id))
}
console.log(`\n✅ smoke suite ที่แนะนำ: ${smoke.length} เทส ` +
            `คลุม ${((covered.size/allKilled.size)*100).toFixed(0)}% ของ mutant ที่ฆ่าได้`)
smoke.forEach((t) => console.log(`   ${t}`))

#!/usr/bin/env node
const GATES = [
  { name: "tsc",         cmd: "pnpm vue-tsc --noEmit" },
  { name: "eslint",      cmd: "pnpm eslint ." },
  { name: "unit",        cmd: "pnpm vitest run test/unit" },
  { name: "integration", cmd: "pnpm vitest run test/integration" },
  { name: "contract",    cmd: "pnpm bruno run" },
  { name: "authz",       cmd: "pnpm vitest run security/authz" },
  { name: "race",        cmd: "pnpm vitest run security/race" },
  { name: "invariant",   cmd: "node monitors/check-invariants.mjs" },
  { name: "e2e",         cmd: "pnpm playwright test --grep @smoke" },
]

const mutants = JSON.parse(readFileSync("/tmp/mutants.json", "utf8"))
const contribution = Object.fromEntries(GATES.map((g) => [g.name, 0]))
const escaped = []

for (const [i, m] of mutants.entries()) {
  execSync(`node mutation/apply.mjs ${i}`)

    catch { caughtBy = gate.name; break }        // ⭐ gate แรกที่จับได้
  }

  if (caughtBy) contribution[caughtBy]++
  else escaped.push(m)                            // ⭐ ผ่านทุก gate

  execSync("git checkout -- .")
}

         COUNT(d.id) AS total_defects

  WHERE d.created_at > NOW() - INTERVAL '180 days'
  GROUP BY m.path, m.mutation_score
  HAVING COUNT(d.id) >= 3
`)

// Spearman rank correlation — ไม่สมมติว่าเป็นเชิงเส้น
function spearman(xs, ys) {
  const rank = (a) => {
    const sorted = [...a].map((v, i) => [v, i]).sort((p, q) => p[0] - q[0])
    const r = new Array(a.length)
    sorted.forEach(([, i], k) => { r[i] = k + 1 })
    return r
  }
  const rx = rank(xs), ry = rank(ys), n = xs.length
  const d2 = rx.reduce((s, v, i) => s + (v - ry[i]) ** 2, 0)
  return 1 - (6 * d2) / (n * (n * n - 1))
}

const scores  = modules.map((m) => m.mutation_score)
const escRate = modules.map((m) => m.escaped / m.total_defects)
const rho = spearman(scores, escRate)

export function stratifiedSample(mutants, budget = 200) {
  const byFile = groupBy(mutants, (m) => m.file)
  const sample = []
  // ⭐ จัดสรรตามความเสี่ยงของไฟล์ ไม่ใช่ตามจำนวน mutant
  const weights = Object.fromEntries(
    Object.keys(byFile).map((f) => [f, riskScoreOf(f)]))
  const totalW = Object.values(weights).reduce((a, b) => a + b, 0)

  for (const [file, ms] of Object.entries(byFile)) {
    const quota = Math.max(1, Math.round((weights[file] / totalW) * budget))
    sample.push(...shuffle(ms).slice(0, quota))
  }
  return sample
}

// ⭐ ต้องรายงานเป็นช่วงความเชื่อมั่น ไม่ใช่ตัวเลขเดียว
export function wilsonInterval(killed, total, z = 1.96) {
  const p = killed / total, n = total
  const denom = 1 + (z * z) / n
  const center = p + (z * z) / (2 * n)
  const spread = z * Math.sqrt((p * (1 - p) + (z * z) / (4 * n)) / n)
  return { low: (center - spread) / denom, high: (center + spread) / denom }
}

import fc from "fast-check"

type Status = "PENDING" | "REVIEWING" | "APPROVED" | "PAID"
             | "FAILED" | "REJECTED" | "CANCELLED"

class WithdrawalModel {
  status: Status = "PENDING"
  paidCount = 0
  // ⭐ ตารางเดียวกับ L1.4 — model คือ spec ที่รันได้
  static VALID: Record<Status, Partial<Record<string, Status>>> = {
    PENDING:   { pickUp: "REVIEWING", cancel: "CANCELLED", timeout: "CANCELLED" },
    REVIEWING: { approve: "APPROVED", reject: "REJECTED" },
    APPROVED:  { payoutSuccess: "PAID", payoutFail: "FAILED" },
    FAILED:    { retry: "APPROVED", reject: "REJECTED" },
    PAID:      {}, REJECTED: {}, CANCELLED: {},
  }
  can(event: string) { return WithdrawalModel.VALID[this.status][event] !== undefined }
  fire(event: string) {
    const next = WithdrawalModel.VALID[this.status][event]
    if (!next) throw new Error(`invalid: ${this.status} --${event}-->`)
    if (next === "PAID") this.paidCount++
    this.status = next
  }

export type MR<I, O> = {
  id: string
  description: string
  /** แปลง input ต้นทางเป็น input ปลายทาง */
  transform: (input: I) => I
  /** ความสัมพันธ์ที่ต้องจริงระหว่าง output ทั้งสอง */
  relation: (source: O, followUp: O, input: I) => boolean
}

export async function checkMR<I, O>(
  mr: MR<I, O>, fn: (i: I) => Promise<O>, inputs: I[]
) {
  const violations = []
  for (const input of inputs) {
    const source = await fn(input)
    const followUp = await fn(mr.transform(input))
    if (!mr.relation(source, followUp, input))
      violations.push({ input, source, followUp })
  }
  expect(violations,
    `⭐ MR ${mr.id} ถูกละเมิด ${violations.length} ครั้ง: ${mr.description}\n` +
    JSON.stringify(violations.slice(0, 3), null, 2)
  ).toEqual([])
}

L5.6Chaos Engineering (ฉบับถูกต้อง)

chaos ≠ สุ่มพังของ

chaos engineering ที่ทำถูกต้องคือการทดลองทางวิทยาศาสตร์ — มีสมมติฐาน มีตัวชี้วัดสภาวะปกติ มีเงื่อนไขยกเลิก และมีขอบเขตความเสียหายที่กำหนดไว้

typescriptchaos/experiment.ts
export type ChaosExperiment = {
  id: string
  hypothesis: string                    // ⭐ ต้องเขียนก่อนรัน
  steadyState: () => Promise<boolean>   // สภาวะปกติวัดยังไง
  inject: () => Promise<() => Promise<void>>  // คืน rollback function
  blastRadius: "single-instance" | "one-endpoint" | "full"
  abortIf: () => Promise<boolean>       // ⭐ เงื่อนไขหยุดทันที
  durationMs: number
}

export async function runChaos(exp: ChaosExperiment) {
  // 1. ยืนยันสภาวะปกติก่อน — ถ้าไม่ปกติอยู่แล้วห้ามรัน
  if (!(await exp.steadyState()))
    return { status: "SKIPPED", reason: "ระบบไม่อยู่ในสภาวะปกติก่อนเริ่ม" }

  const rollback = await exp.inject()
  const started = Date.now()
  let aborted = false

  try {
    while (Date.now() - started < exp.durationMs) {
      if (await exp.abortIf()) { aborted = true; break }
      await sleep(2000)
    }
    const steadyDuring = await exp.steadyState()
    return {
      status: aborted ? "ABORTED" : steadyDuring ? "HYPOTHESIS_HELD" : "HYPOTHESIS_FAILED",
      // ⭐ HYPOTHESIS_FAILED = ค้นพบ ไม่ใช่ความล้มเหลว
    }
  } finally {
    await rollback()
    // ⭐ recovery stage — สำคัญพอกับตัวการทดลอง
    await sleep(30_000)
    const recovered = await exp.steadyState()
    if (!recovered) await alertTelegram({
      severity: "critical",
      title: `❌ ระบบไม่ฟื้นตัวหลัง chaos ${exp.id}`,
    })
  }
}

แก้ไขจากการทบทวน — ใช้ McNemar บน pass rate ที่ binarize แล้ว

ฉบับแรกเสนอให้รันแต่ละเคส 20 ครั้ง → คำนวณ pass rate → บีบเป็น pass/fail (เช่น “ผ่านถ้า ≥ 80%”) → แล้วใช้ McNemar test เปรียบเทียบสองเวอร์ชัน

ปัญหา 3 ข้อ

  1. ทิ้งข้อมูลที่อุตส่าห์เก็บมา — รัน 20 ครั้งเพื่อได้ความละเอียด แล้วบีบเหลือ 1 บิต เคสที่ 85% กับ 100% ถูกนับเท่ากัน
  2. ความแปรปรวนสองระดับถูกละเลย — มีทั้งความแปรปรวนระหว่างเคส (บางเคสยากกว่า) และภายในเคส (การสุ่มของ LLM) McNemar สมมติว่ามีแค่ระดับเดียว
  3. threshold ที่ตั้งเอง — 80% มาจากไหน? เปลี่ยนเป็น 75% ผลอาจกลับข้าง

วิธีที่ถูก: hierarchical (cluster) bootstrap — resample ทั้งสองระดับ แล้วดูช่วงความเชื่อมั่นของผลต่าง

typescripteval/stats/hierarchical-bootstrap.ts
export type CaseRuns = {
  caseId: string
  /** คะแนน 0.0–1.0 ต่อการรันหนึ่งครั้ง (ไม่ binarize) */
  scoresA: number[]
  scoresB: number[]
}

function mean(xs: number[]) { return xs.reduce((a, b) => a + b, 0) / xs.length }
function pick<T>(xs: T[]) { return xs[Math.floor(Math.random() * xs.length)] }

/**
 * ⭐ resample 2 ระดับ:
 *   ระดับ 1 — สุ่มเลือก "เคส" ใหม่พร้อมแทนที่ (คุมความแปรปรวนระหว่างเคส)
 *   ระดับ 2 — ภายในแต่ละเคสที่เลือกมา สุ่ม "การรัน" ใหม่พร้อมแทนที่
 */
export function hierarchicalBootstrap(data: CaseRuns[], iterations = 10_000) {
  const deltas: number[] = []

  for (let i = 0; i < iterations; i++) {
    const sampledCases = Array.from({ length: data.length }, () => pick(data))

    const caseDeltas = sampledCases.map((c) => {
      const a = Array.from({ length: c.scoresA.length }, () => pick(c.scoresA))
      const b = Array.from({ length: c.scoresB.length }, () => pick(c.scoresB))
      return mean(b) - mean(a)
    })
    deltas.push(mean(caseDeltas))
  }

  deltas.sort((x, y) => x - y)
  const q = (p: number) => deltas[Math.floor(p * deltas.length)]

  const observed = mean(data.map((c) => mean(c.scoresB) - mean(c.scoresA)))
  return {
    delta: observed,
    ci95: [q(0.025), q(0.975)] as [number, number],
    // ⭐ ความน่าจะเป็นที่ B ดีกว่า A จริง
    pBetter: deltas.filter((d) => d > 0).length / deltas.length,
  }
}

// ⭐ scoring ที่ไม่ binarize
function scoreCase(output: AgentOutput, c: EvalCase): number {
  // mustNotHave ละเมิด = 0 ทันที
  if (c.expectations.mustNotHave.some((p) => matches(output, p))) return 0
  // mustHave ต้องครบ ถ้าขาด = 0
  const mustOk = c.expectations.mustHave.every((p) => matches(output, p))
  if (!mustOk) return 0
  // shouldHave ให้คะแนนบางส่วน
  const should = c.expectations.shouldHave
  const hit = should.filter((p) => matches(output, p)).length
  return 0.6 + 0.4 * (should.length ? hit / should.length : 1)
}

L5.7.4 — Calibration ของ judge ⭐⭐

แก้ไขจากการทบทวน — ใช้ Cohen's kappa กับข้อมูลที่ไม่เหมาะ

ฉบับแรกเสนอให้วัดความสอดคล้องระหว่าง judge กับคนด้วย Cohen's kappa และตั้งเกณฑ์ κ > 0.6 — มีปัญหา 3 ข้อ

ปัญหาผลที่เกิด
Kappa paradoxถ้า calibration set เอียง (90% เป็นเคสดี) → agreement 94% แต่ κ = 0.37 → “ล้มเหลว” ทั้งที่ judge แม่นมาก
κ ไม่เหมาะกับ ordinalคะแนน 1–5 มีลำดับ — ให้ 4 แทน 5 ต่างจากให้ 1 แทน 5 มาก แต่ κ ธรรมดานับเป็น “ไม่ตรงกัน” เท่ากัน
ไม่บอกทิศทางjudge ใจดีเกินไปหรือเข้มเกินไป? κ บอกไม่ได้

วิธีที่ถูก: ใช้ 4 ตัวชี้วัดคู่กัน — QWK · MAE · bias · agreement ดิบ และ calibration set ต้อง stratified (มีทุกระดับคะแนนใกล้เคียงกัน)

type Pair = { human: number; judge: number }   // คะแนน 1-5

/** Quadratic Weighted Kappa — เหมาะกับ ordinal, ลงโทษตามระยะห่างกำลังสอง */
export function qwk(pairs: Pair[], levels = 5): number {
  const n = pairs.length
  const O = Array.from({ length: levels }, () => new Array(levels).fill(0))
  for (const p of pairs) O[p.human - 1][p.judge - 1]++

  const humanHist = new Array(levels).fill(0)
  const judgeHist = new Array(levels).fill(0)
  for (const p of pairs) { humanHist[p.human - 1]++; judgeHist[p.judge - 1]++ }

  let num = 0, den = 0
  for (let i = 0; i < levels; i++)
    for (let j = 0; j < levels; j++) {
      const w = ((i - j) ** 2) / ((levels - 1) ** 2)
      const E = (humanHist[i] * judgeHist[j]) / n
      num += w * O[i][j]
      den += w * E
    }
  return 1 - num / den
}

/** ค่าเฉลี่ยความคลาดเคลื่อนสัมบูรณ์ — ตีความง่ายที่สุด */
export function mae(pairs: Pair[]): number {
  return pairs.reduce((s, p) => s + Math.abs(p.human - p.judge), 0) / pairs.length
}

/** ⭐ bias — บวก = judge ใจดีกว่าคน, ลบ = เข้มกว่าคน */
export function bias(pairs: Pair[]): number {
  return pairs.reduce((s, p) => s + (p.judge - p.human), 0) / pairs.length
}

/** PABAK — kappa ที่ปรับ prevalence แล้ว ใช้ตรวจว่าเจอ paradox มั้ย */
export function pabak(pairs: Pair[]): number {
  const agree = pairs.filter((p) => p.human === p.judge).length / pairs.length
  return 2 * agree - 1
}

export function calibrationReport(pairs: Pair[]) {
  const agreement = pairs.filter((p) => p.human === p.judge).length / pairs.length
  const within1  = pairs.filter((p) => Math.abs(p.human - p.judge) <= 1).length / pairs.length
  const r = {
    n: pairs.length,
    agreement, within1,
    qwk: qwk(pairs), mae: mae(pairs), bias: bias(pairs), pabak: pabak(pairs),
  }

  console.log(`n=${r.n} · agreement=${(agreement*100).toFixed(0)}% · ` +
              `within±1=${(within1*100).toFixed(0)}%`)
  console.log(`QWK=${r.qwk.toFixed(3)} · MAE=${r.mae.toFixed(2)} · ` +
              `bias=${r.bias > 0 ? "+" : ""}${r.bias.toFixed(2)} · PABAK=${r.pabak.toFixed(3)}`)

  // ⭐ ตรวจ kappa paradox
  if (agreement > 0.85 && r.qwk < 0.5)
    console.log("⚠️  kappa paradox: agreement สูงแต่ QWK ต่ำ — calibration set เอียง ต้อง stratify ใหม่")

  // ⭐ เกณฑ์ที่ปฏิบัติได้จริง
  const verdict =
    r.mae <= 0.5 && Math.abs(r.bias) <= 0.3 && within1 >= 0.9 ? "ใช้ได้"
    : Math.abs(r.bias) > 0.5 ? `ต้องปรับ rubric — judge ${r.bias > 0 ? "ใจดี" : "เข้ม"}เกินไป`
    : "ต้องปรับ rubric — คลาดเคลื่อนสูง"

  console.log(`→ ${verdict}`)
  return { ...r, verdict }
}

export async function measureContextEffect(opts: {
  contextFile: string          // "fault-attacks.md"
  metric: "first_pass_rate" | "human_reject_rate" | "escaped_defect_rate"
  minSample?: number
}) {
  const runs = await db.query(`
    SELECT * FROM agent_runs
    WHERE stage IN ('S3','S4') AND created_at > NOW() - INTERVAL '180 days'
  `)

  const withFile = runs.filter((r) => r.context_files.includes(opts.contextFile))
  const without  = runs.filter((r) => !r.context_files.includes(opts.contextFile))

  const min = opts.minSample ?? 40
  if (withFile.length < min || without.length < min)
    return { verdict: "ข้อมูลไม่พอ", nWith: withFile.length, nWithout: without.length }

  // ⭐ ต้องคุมตัวแปรกวน: ticket ที่ยากกว่ามักมาทีหลัง (พร้อมกับ context ที่ดีขึ้น)
  const stratify = (rs: any[]) => groupBy(rs, (r) => r.tier)
  const byTierWith = stratify(withFile), byTierWithout = stratify(without)

  const results: any[] = []
  for (const tier of ["T1", "T2", "T3"]) {
    const a = byTierWithout[tier] ?? [], b = byTierWith[tier] ?? []
    if (a.length < 15 || b.length < 15) continue
    results.push({
      tier,
      without: rate(a, opts.metric),
      with: rate(b, opts.metric),
      delta: rate(b, opts.metric) - rate(a, opts.metric),
      n: { without: a.length, with: b.length },
    })
  }

  console.table(results)
  // ⭐ ถ้าผลไปคนละทางในแต่ละ tier = ยังสรุปไม่ได้
  const consistent = results.every((r) => Math.sign(r.delta) === Math.sign(results[0].delta))
  return { results, consistent,
    verdict: !consistent ? "ผลไม่สอดคล้องกันข้าม tier — ยังสรุปไม่ได้"
      : results[0].delta > 0.05 ? `${opts.contextFile} ช่วยจริง`
      : "ไม่เห็นผลชัดเจน — พิจารณาเอาออกเพื่อลด context" }

export type ToolCall = {
  tool: "read" | "write" | "bash" | "search" | "edit"
  target: string
  ts: number
  ok: boolean
}

export function analyzeTrajectory(calls: ToolCall[], stage: string) {
  const signals: string[] = []

  // 1. อ่านไฟล์เดิมซ้ำหลายครั้ง = context ไม่พอ หรือหลงทาง
  const readCounts = countBy(calls.filter((c) => c.tool === "read"), (c) => c.target)
  const rereads = Object.entries(readCounts).filter(([, n]) => n >= 3)
  if (rereads.length)
    signals.push(`อ่านซ้ำ ≥3 ครั้ง: ${rereads.map(([f]) => f).join(", ")}`)

  // 2. เขียนแล้วลบแล้วเขียนใหม่ = backtrack
  const writes = calls.filter((c) => c.tool === "write" || c.tool === "edit")
  const writeCounts = countBy(writes, (c) => c.target)
  const churn = Object.entries(writeCounts).filter(([, n]) => n >= 4)
  if (churn.length)
    signals.push(`แก้ไฟล์เดิม ≥4 ครั้ง (backtrack): ${churn.map(([f]) => f).join(", ")}`)

  // 3. ⭐ แตะไฟล์นอกขอบเขตของ stage
  const FORBIDDEN: Record<string, RegExp[]> = {
    S3: [/^src\//, /^server\//],              // S3 เขียนแต่เทส
    S4: [/\.spec\.ts$/, /\.test\.ts$/],       // S4 ห้ามแตะเทส
  }
  const violations = writes.filter((c) =>
    (FORBIDDEN[stage] ?? []).some((re) => re.test(c.target)))
  if (violations.length)
    signals.push(`⭐ แตะไฟล์นอกขอบเขต: ${violations.map((v) => v.target).join(", ")}`)

  // 4. ⭐ pattern ที่น่าสงสัย
  const bash = calls.filter((c) => c.tool === "bash").map((c) => c.target)
  const SUSPICIOUS = [
    { re: /curl|wget|nc\s/, why: "พยายามเข้าถึงเครือข่ายภายนอก" },
    { re: /\.env|credentials|\.ssh|id_rsa/, why: "พยายามอ่าน credential" },
    { re: /--update-snapshot/, why: "พยายาม update snapshot" },
    { re: /git\s+push|git\s+reset\s+--hard/, why: "คำสั่ง git ที่อันตราย" },
    { re: /rm\s+-rf/, why: "ลบไฟล์แบบ recursive" },
  ]
  for (const s of SUSPICIOUS)
    if (bash.some((b) => s.re.test(b)))
      signals.push(`🔴 ${s.why}`)

  // 5. อัตราส่วนที่บอกประสิทธิภาพ
  const efficiency = writes.length / Math.max(calls.length, 1)

  return {
    totalCalls: calls.length,
    efficiency,
    signals,
    // ⭐ ต้องแดงถ้ามีสัญญาณสีแดง
    verdict: signals.some((s) => s.startsWith("🔴")) ? "BLOCK"
           : signals.length >= 3 ? "REVIEW" : "OK",
  }
}

const INJECTION_PATTERNS = [
  /ignore\s+(all\s+)?(previous|prior|above)\s+instructions?/i,
  /disregard\s+(the\s+)?(above|previous)/i,
  /new\s+(task|instruction|objective)\s*:/i,
  /^\s*system\s*:/im,
  /<\|?(im_start|system|assistant)\|?>/i,
  /\[\/?(INST|SYS)\]/i,
  /you\s+are\s+now\s+/i,

  /คำสั่งใหม่\s*:/,
  /เพิกเฉย(ต่อ)?คำสั่ง/,
]

export function wrapUntrusted(content: string, source: string) {
  const findings = INJECTION_PATTERNS.filter((re) => re.test(content))

  // ชั้น 3: sanitize marker ที่อันตราย
  const cleaned = content
    .replace(/<\|[^|]*\|>/g, "[ตัดออก]")
    .replace(/^\s*system\s*:/gim, "[ตัดออก]:")
    .replace(/\[\/?(INST|SYS)\]/gi, "[ตัดออก]")

  // ชั้น 1: boundary ที่ชัดเจน + คำเตือน
  const wrapped = [
    `<untrusted_content source="${source}">`,

    `---`,
    cleaned,
    `---`,
    `</untrusted_content>`,
  ].join("\n")

  return { wrapped, suspicious: findings.length > 0, patternCount: findings.length }
}

typescriptanalyze/spc.ts — Western Electric rules
export function controlLimits(baseline: number[]) {
  const n = baseline.length
  const mean = baseline.reduce((a, b) => a + b, 0) / n
  const sd = Math.sqrt(baseline.reduce((s, x) => s + (x - mean) ** 2, 0) / (n - 1))
  return { mean, sd, ucl: mean + 3 * sd, lcl: mean - 3 * sd }
}

/** Western Electric rules — จับ pattern ที่ threshold เดี่ยวจับไม่ได้ */
export function westernElectric(series: number[], lim: ReturnType<typeof controlLimits>) {
  const { mean, sd } = lim
  const z = series.map((x) => (x - mean) / sd)
  const alerts: string[] = []
  const last = (n: number) => z.slice(-n)

  // R1: 1 จุดเกิน 3σ
  if (Math.abs(z.at(-1)!) > 3) alerts.push("R1: จุดล่าสุดเกิน 3σ — เหตุการณ์เฉียบพลัน")

  // R2: 2 จาก 3 จุดล่าสุด เกิน 2σ ฝั่งเดียวกัน
  const l3 = last(3)
  if (l3.length === 3) {
    for (const sign of [1, -1])
      if (l3.filter((v) => v * sign > 2).length >= 2)
        alerts.push("R2: 2/3 จุดเกิน 2σ ฝั่งเดียวกัน")
  }

  // R3: 4 จาก 5 จุดล่าสุด เกิน 1σ ฝั่งเดียวกัน
  const l5 = last(5)
  if (l5.length === 5) {
    for (const sign of [1, -1])
      if (l5.filter((v) => v * sign > 1).length >= 4)
        alerts.push("R3: 4/5 จุดเกิน 1σ ฝั่งเดียวกัน")
  }

  // ⭐ R4: 8 จุดติดกันอยู่ฝั่งเดียวของค่ากลาง — จับ drift ที่คืบช้าๆ
  const l8 = last(8)
  if (l8.length === 8 && (l8.every((v) => v > 0) || l8.every((v) => v < 0)))
    alerts.push("⭐ R4: 8 จุดติดกันฝั่งเดียว — มีการเปลี่ยนแปลงเชิงระบบ (model drift?)")

  // R5: 6 จุดติดกันเพิ่มขึ้นหรือลดลงต่อเนื่อง
  const l6 = last(6)
  if (l6.length === 6) {
    const inc = l6.every((v, i) => i === 0 || v > l6[i - 1])
    const dec = l6.every((v, i) => i === 0 || v < l6[i - 1])
    if (inc || dec) alerts.push(`R5: 6 จุด${inc ? "เพิ่ม" : "ลด"}ต่อเนื่อง — trend`)
  }

  return alerts
}

/** CUSUM — จับการเลื่อนเล็กๆ ที่ต่อเนื่อง ไวกว่า control chart */
export function cusum(series: number[], target: number, k = 0.5, h = 5) {
  let sHi = 0, sLo = 0
  const signals: number[] = []
  series.forEach((x, i) => {
    sHi = Math.max(0, sHi + (x - target) - k)
    sLo = Math.max(0, sLo - (x - target) - k)
    if (sHi > h || sLo > h) signals.push(i)
  })
  return { sHi, sLo, signals, drifted: sHi > h || sLo > h }
}

export type Factor = { name: string; values: string[] }
export type Constraint = (row: Record<string, string>) => boolean

export function generateCoveringArray(opts: {
  factors: Factor[]
  strength?: number                        // t-way (default 2)
  constraints?: Constraint[]
  seedRows?: Record<string, string>[]      // ⭐ combination ที่ต้องมีแน่นอน
}): Record<string, string>[] {
  const t = opts.strength ?? 2
  const constraints = opts.constraints ?? []
  const valid = (row: Record<string, string>) => constraints.every((c) => c(row))

  // 1. สร้าง t-tuple ที่ต้องคลุมทั้งหมด (เฉพาะที่ผ่าน constraint)
  const tuples = new Set<string>()
  for (const combo of kCombinations(opts.factors, t))
    for (const values of cartesian(...combo.map((f) => f.values))) {
      const partial = Object.fromEntries(combo.map((f, i) => [f.name, values[i]]))
      if (constraints.every((c) => c({ ...partial } as any) !== false))
        tuples.add(JSON.stringify(partial))
    }

  const rows: Record<string, string>[] = []

  // 2. ⭐ ใส่ seed row ก่อน — combination ที่ risk สูงต้องมีแน่นอน
  for (const seed of opts.seedRows ?? []) {
    if (!valid(seed)) throw new Error(`seed row ละเมิด constraint: ${JSON.stringify(seed)}`)
    rows.push(seed)
    removeCovered(tuples, seed, opts.factors, t)
  }

  // 3. greedy (IPOG-like) — แต่ละรอบเลือกแถวที่คลุม tuple ใหม่ได้มากสุด
  let guard = 0
  while (tuples.size > 0 && guard++ < 10_000) {
    let best: Record<string, string> | null = null
    let bestGain = -1

    for (let attempt = 0; attempt < 200; attempt++) {
      const cand = greedyCandidate(opts.factors, tuples, constraints)
      if (!cand || !valid(cand)) continue
      const gain = countNewlyCovered(tuples, cand, opts.factors, t)
      if (gain > bestGain) { bestGain = gain; best = cand }
    }
    if (!best || bestGain === 0) break
    rows.push(best)
    removeCovered(tuples, best, opts.factors, t)
  }

  // 4. ⭐ ต้องพิสูจน์ว่าคลุมครบ ไม่ใช่เชื่อว่าครบ
  if (tuples.size > 0)
    throw new Error(`คลุมไม่ครบ เหลือ ${tuples.size} tuple:\n` +
                    [...tuples].slice(0, 5).join("\n"))

  return rows
}

✎แบบฝึกหัด L5 — เรียงตามผลตอบแทน

  1. Mutation บนโค้ดเงิน ⭐⭐ — รัน Stryker กับ utils/money + utils/bonus → triage M1–M5 → แก้เฉพาะ M2 ที่ cluster ใหญ่สุด 5 จุด
  2. Custom mutator ⭐⭐ — implement AwaitRemoval + IdempotencyBypass + DoAdminRemoval → รัน → ถ้ารอดแม้ตัวเดียวคือช่องโหว่จริง
  3. Gate mutation ⭐⭐ — รัน mutant ผ่าน gate chain ทั้งหมด → ได้ escape profile → ใช้เป็นรายการงานว่าจะสร้าง gate อะไรต่อ
  4. Injection eval ⭐⭐ — สร้าง 5 เคส → รันใน sandbox → ถ้าสำเร็จแม้เคสเดียว หยุดทุกอย่างแล้วทำ capability restriction
  5. Capability restriction ⭐⭐ — implement STAGE_CAPABILITIES → ทดสอบด้วยการให้ agent พยายามอ่าน .env
  6. Property test ของ ledger — เขียนชั้น model-only 5,000 runs ก่อน → แล้วค่อยทำชั้น real-system 15 runs
  7. Differential fuzzing ⭐ — เก็บ calcBonus เวอร์ชันเก่าไว้ → ให้ agent refactor → รัน 20,000 runs
  8. Model-based withdrawal — implement InvalidCommand ที่ยิงทุกสถานะ → ดูว่าเจอ transition ที่ไม่มีใครป้องกันกี่ตัว
  9. Eval dataset ⭐ — สร้าง golden 30 เคส + ตั้ง audit repo แยกสำหรับ held-out
  10. Judge calibration ⭐ — สร้าง stratified calibration set (20 เคส × 5 ระดับ) → วัด QWK/MAE/bias → ปรับ rubric จนกว่า MAE ≤ 0.5
  11. Model fingerprint — รัน 6 probe → บันทึก baseline → ตั้ง cron รายวัน
  12. Context effect ⭐ — วัดว่า fault-attacks.md ช่วยจริงมั้ยด้วยข้อมูลที่มีอยู่แล้ว → ถ้าไม่ช่วย พิจารณาเอาออกเพื่อลด context
  13. SPC บน canary eval — เก็บคะแนนรายวัน 20 จุด → ตั้ง control chart → เปิด R4 alert
